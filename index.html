# Function for converting CI's to SD

me_to_sd <- function(me, n, ci_level = 0.95) {
  alpha <- 1 - ci_level
  tcrit <- qt(1 - alpha/2, df = n - 1)
  me * sqrt(n) / tcrit
}

# Function for converting SE to SD

se_to_sd <- function(se, n) {
  # se: standard error
  # n: sample size
  # return: standard deviation
  sd_value <- se * sqrt(n)
  return(sd_value)
}
